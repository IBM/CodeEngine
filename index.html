async function subscribeToStorage(
  projectId: string,
  appName: string,
  bucketName: string
) {
  const subscription = await 
    codeEngineClient.createSubscriptionCos({
      projectId,
      name: `cos-trigger-${bucketName}`,
      destination: { 
        name: appName, 
        type: "app" 
      },
      source: { bucket: bucketName },
    });
}

async function buildFromSource(
  projectId: string,
  repoUrl: string,
  outputImage: string
) {
  // create build config
  await codeEngineClient.createBuild({
    projectId,
    name: "my-build",
    outputImage,
    sourceUrl: repoUrl,
    strategyType: "dockerfile",
  });
}
